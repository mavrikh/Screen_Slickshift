# Slickshift mainline dependencies
# Python 3.10+ required.

PyQt6>=6.6

# pyautogui is used for passive cursor position polling (pyautogui.position())
# and injection (pyautogui.moveTo/moveRel/mouseDown/mouseUp/scroll/hscroll/keyDown/keyUp).
#
# PROVISIONAL -- production capture will replace pyautogui.position() with
# CGEventTap on macOS and RawInput on Windows (see capture/mouse_capture.py stub).
# The injection surface (moveTo, mouseDown, etc.) may also be replaced with
# direct platform API calls at that time.
pyautogui>=0.9.54

# pynput is used for cross-platform mouse button and scroll event capture.
# Runs a listener in a background thread. On macOS requires Accessibility permission
# (same grant as pyautogui). On Windows no extra permission is needed.
pynput>=1.7

# pyobjc-framework-Quartz drives the macOS exclusive-mode capture path
# (slickshift/capture/mac_event_tap.py). Installs a CGEventTap at
# kCGHIDEventTap so mouse events can be suppressed before the OS cursor
# moves. Not imported on Windows/Linux.
pyobjc-framework-Quartz>=10.0; sys_platform == "darwin"

# websockets is used by slickshift.browser_agent.server to run the local
# WebSocket server that the Chrome extension connects to over loopback.
# The modern asyncio API (websockets.asyncio.server) requires version >= 14.
websockets>=14.0

# pyperclip provides cross-platform clipboard read/write. Used by
# slickshift.browser_agent.clipboard to read the host clipboard before
# dispatching open_url or paste_text to the connected browser extension.
pyperclip>=1.8
